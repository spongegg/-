# -*- coding: utf-8 -*- import os import re import shutil import tkinter as tk from collections import defaultdict from tkinter import filedialog, messagebox, simpledialog import pysubs2 # 字幕重命名类 class SubtitleRenamer: """负责字幕重命名功能的类""" def __init__(self, root): self.root = root def normalize_ep(self, match): s, e = match.groups() return f"S{int(s):02d}E{int(e):02d}" def detect_lang(self, filename): name = filename.lower() zh = re.search(r'(?:\b|_|\.)?(zh|chs|cht|chi|cn|gb|big5|中文)(?:\b|_|\.)?', name) en = re.search(r'(?:\b|_|\.)?(en|eng|英文)(?:\b|_|\.)?', name) if zh and en: return "zh&en" elif zh: return "zh" elif en: return "en" else: return "" def run(self): self.rename_window = tk.Toplevel(self.root) self.rename_window.title("字幕重命名") self.rename_window.geometry("600x620") self.rename_window.minsize(600, 620) # 选择视频文件 tk.Label(self.rename_window, text="视频文件:").pack(pady=5) self.video_listbox = tk.Listbox(self.rename_window, height=5) self.video_listbox.pack(fill=tk.X, padx=10) tk.Button(self.rename_window, text="选择视频文件", command=self.select_videos).pack(pady=5) # 选择字幕文件 tk.Label(self.rename_window, text="字幕文件:").pack(pady=5) self.sub_listbox = tk.Listbox(self.rename_window, height=5) self.sub_listbox.pack(fill=tk.X, padx=10) tk.Button(self.rename_window, text="选择字幕文件", command=self.select_subs).pack(pady=5) # 正则表达式输入 tk.Label(self.rename_window, text="正则表达式 (留空则顺序匹配):").pack(pady=5) self.regex_entry = tk.Entry(self.rename_window) self.regex_entry.pack(fill=tk.X, padx=10) self.regex_entry.insert(0, r"[Ss](\d{1,2})[Ee](\d{1,2})") # 匹配结果显示 tk.Label(self.rename_window, text="匹配结果:").pack(pady=5) self.result_text = tk.Text(self.rename_window, height=10) self.result_text.pack(fill=tk.BOTH, expand=True, padx=10, pady=5) self.result_text.tag_config("video", foreground="green") self.result_text.tag_config("sub", foreground="blue") self.result_text.tag_config("ep", foreground="orange") # 按钮 button_frame = tk.Frame(self.rename_window) button_frame.pack(fill=tk.X, padx=10, pady=10) tk.Button(button_frame, text="匹配", command=self.match_files, width=12).pack(side=tk.LEFT) tk.Button(button_frame, text="确认重命名", command=self.confirm_rename, width=12).pack(side=tk.RIGHT) self.root.wait_window(self.rename_window) self.videos = [] self.subs = [] self.pairs = [] def select_videos(self): videos = filedialog.askopenfilenames( title="请选择视频文件", initialdir=os.path.expanduser("~"), filetypes=[("视频文件", "*.mp4 *.avi *.mkv *.mov")] ) if videos: self.videos = videos self.video_listbox.delete(0, tk.END) for vid in videos: self.video_listbox.insert(tk.END, os.path.basename(vid)) if self.subs: self.match_files() def select_subs(self): subs = filedialog.askopenfilenames( title="请选择字幕文件", initialdir=os.path.expanduser("~"), filetypes=[("字幕文件", "*.srt *.ass *.vtt")] ) if subs: self.subs = subs self.sub_listbox.delete(0, tk.END) for sub in subs: self.sub_listbox.insert(tk.END, os.path.basename(sub)) if self.videos: self.match_files() def match_files(self): if not self.videos or not self.subs: messagebox.showwarning("警告", "请先选择视频和字幕文件。") return regex_pattern = self.regex_entry.get().strip() pattern = None if regex_pattern: try: pattern = re.compile(regex_pattern, re.I) except re.error as err: messagebox.showerror("正则错误", f"无效正则表达式: {err}") return ep_pattern = re.compile(r'[Ss](\d{1,2})[Ee](\d{1,2})') # ---------- 预处理 ---------- sub_map = defaultdict(list) # SxxEyy -> [sub,...] sub_bases = {} # sub -> basename unmatched_subs = set(self.subs) for sub in self.subs: base = os.path.splitext(os.path.basename(sub))[0] sub_bases[sub] = base.lower() m = ep_pattern.search(base) if m: key = self.normalize_ep(m) sub_map[key].append(sub) self.pairs = [] unmatched_videos = [] # ---------- 匹配 ---------- for vid in self.videos: vbase = os.path.splitext(os.path.basename(vid))[0] vbase_lower = vbase.lower() candidate = None # 1️⃣ SxxEyy 精确匹配（最快最关键） m = ep_pattern.search(vbase) if m: key = self.normalize_ep(m) if key in sub_map and sub_map[key]: candidate = sub_map[key].pop(0) # 2️⃣ 完全匹配 if not candidate: for s, b in sub_bases.items(): if b == vbase_lower: candidate = s break # 3️⃣ 包含匹配 if not candidate: for s, b in sub_bases.items(): if vbase_lower in b or b in vbase_lower: candidate = s break # 4️⃣ 自定义正则 if not candidate and pattern: for s, b in sub_bases.items(): if pattern.search(b): candidate = s break # 5️⃣ 顺序匹配（仅在未填写正则时） if not candidate and pattern is None: for s in self.subs: if s in unmatched_subs: candidate = s break # 记录结果 if candidate: self.pairs.append((vid, candidate)) unmatched_subs.discard(candidate) else: unmatched_videos.append(vid) # ---------- 输出 ---------- self.result_text.delete(1.0, tk.END) self.result_text.insert(tk.END, "匹配结果：\n\n") for i, (v, s) in enumerate(self.pairs, 1): vname = os.path.basename(v) sname = os.path.basename(s) ep = re.search(r'[Ss]\d{1,2}[Ee]\d{1,2}', vname) ep_str = ep.group().upper() if ep else "UNKNOWN" self.result_text.insert(tk.END, f"[{i:02d}] ", "ep") self.result_text.insert(tk.END, f"{ep_str}\n", "ep") self.result_text.insert(tk.END, " 视频: ", "video") self.result_text.insert(tk.END, f"{vname}\n") self.result_text.insert(tk.END, " 字幕: ", "sub") self.result_text.insert(tk.END, f"{sname}\n\n") if unmatched_videos: self.result_text.insert(tk.END, "未匹配视频:\n") for v in unmatched_videos: self.result_text.insert(tk.END, f"- {os.path.basename(v)}\n") self.result_text.insert(tk.END, "\n") if unmatched_subs: self.result_text.insert(tk.END, "未匹配字幕:\n") for s in unmatched_subs: self.result_text.insert(tk.END, f"- {os.path.basename(s)}\n") def confirm_rename(self): if not self.pairs: messagebox.showwarning("警告", "没有匹配对，无法重命名。") return if not messagebox.askyesno( "确认重命名", f"已匹配 {len(self.pairs)} 对文件。\n请确认是否继续将字幕重命名并移动到对应视频目录？" ): return errors = [] dirs_seen = set() for vid, sub in self.pairs: base = os.path.splitext(os.path.basename(vid))[0] ext = os.path.splitext(sub)[1] lang = self.detect_lang(os.path.basename(sub)) if lang: newname = f"{base}.{lang}{ext}" else: newname = f"{base}.unknown{ext}" dest = os.path.join(os.path.dirname(vid), newname) counter = 1 while os.path.exists(dest): suffix = lang if lang else "unknown" newname = f"{base}.{suffix}.{counter}{ext}" dest = os.path.join(os.path.dirname(vid), newname) counter += 1 if os.path.abspath(sub) == os.path.abspath(dest): continue try: shutil.move(sub, dest) except Exception as e: errors.append((sub, str(e))) dirs_seen.add(os.path.dirname(sub)) # cleanup for folder in dirs_seen: try: if os.path.isdir(folder) and not os.listdir(folder): os.rmdir(folder) except Exception as e: pass if errors: error_info = "以下字幕未能重命名:\n" for f, err in errors: error_info += f"{os.path.basename(f)}: {err}\n" messagebox.showerror("处理错误", error_info) else: messagebox.showinfo("处理完成", f"成功处理 {len(self.pairs)} 个字幕文件。") self.rename_window.destroy() class SubtitleShifter: """负责调整字幕时间轴功能的类""" def __init__(self, root): self.root = root def adjust_subtitle_file(self, file_path, offset_ms): temp_file_path = file_path + ".tmp" try: subs = None _, ext = os.path.splitext(file_path) forced_format = ext[1:] if ext else None if not forced_format in ('srt', 'ass', 'ssa', 'vtt'): raise ValueError(f"不支持的文件扩展名: '{ext}'") try: subs = pysubs2.load(file_path) except Exception: try: subs = pysubs2.load(file_path, encoding="utf-16", format_=forced_format) except Exception: subs = pysubs2.load(file_path, encoding="utf-8", format_=forced_format) subs.shift(ms=offset_ms) subs.save(temp_file_path, format_=forced_format) os.remove(file_path) os.rename(temp_file_path, file_path) return True, file_path except Exception as e: if os.path.exists(temp_file_path): os.remove(temp_file_path) error_message = f"在处理文件 '{os.path.basename(file_path)}' 时出错:\n{e}\n\n这很可能意味着字幕文件本身已损坏或格式不正确。" messagebox.showerror("处理错误", error_message) return False, str(e) def run(self): offset_ms_val = simpledialog.askinteger( "输入偏移量", "请输入本次批量处理的时间偏移量（毫秒）。\n正数使字幕延后，负数使字幕提前。", minvalue=None ) if offset_ms_val is None: messagebox.showinfo("提示", "操作已取消。") return messagebox.showinfo("提示", f"已设定延迟: {offset_ms_val} 毫秒。现在请选择要处理的文件...") subtitle_files = filedialog.askopenfilenames( title=f"请选择要应用 [{offset_ms_val} ms] 延迟的字幕档", initialdir=os.path.expanduser("~"), filetypes=[("字幕档", "*.srt *.ass *.vtt"), ("所有档案", "*.*")] ) if not subtitle_files: messagebox.showinfo("提示", "您没有选择任何档案。") return if not messagebox.askyesno( "警告：将覆盖原始文件", f"您选择了 {len(subtitle_files)} 个文件。\n" f"将对所有这些文件应用【{offset_ms_val} 毫秒】的延迟调整。\n" "此操作将直接修改并覆盖原始文件，无法撤销。\n\n" "确定要继续吗？" ): messagebox.showinfo("提示", "操作已由用户取消。") return processed_count = 0 for s_file in subtitle_files: success, result_path = self.adjust_subtitle_file(s_file, offset_ms_val) if success: processed_count += 1 else: messagebox.showerror("处理错误", f"调整失败: {result_path}") messagebox.showinfo( "处理完成", f"本批次任务完成！\n\n总共选择了 {len(subtitle_files)} 个档案。\n" f"成功处理了 {processed_count} 个档案。" ) class SubtitleToolboxGUI(tk.Tk): """主程序GUI，负责显示菜单和调度功能""" def __init__(self): super().__init__() self.title("字幕工具箱") self.geometry("400x200") self.resizable(False, False) # 创建按钮 tk.Button(self, text="字幕重命名", command=self.rename_subtitles, width=20, height=2).pack(pady=10)
